# @weight 1

busybox's ar (archival/ar.c, libarchive's unpack_ar_archive and
get_header_ar), with long names and r on.  Every expectation is busybox
1.36.1's own output, built from its source with CONFIG_AR on (its defconfig
leaves it off) and run as an ordinary user, less the banner line its usage
text starts with.  A `|` marks the end of each line written to standard
output, and a verbose listing's time is shown as TIME, since it is the
timezone's.  `after:` is what a command run after shows: an archive with each
header's owner as OWNER, or files with their modes and bytes.

## the fixtures

### the archives and files, and a run of ar in a fresh directory

`mk.sh` makes them, every time dated 2020-01-02 03:04:05 UTC.  `a.a` holds
`hello.txt`, `odd` (three bytes, padded to an even byte) and an empty `z.c`;
`gnu.a` a symbol table, a table of long names and two members named from it;
the rest are archives each wrong in one way, an empty file, text, and files
to add.  Each case runs in `/tmp/x-cu-ar/w`, made afresh: PRE runs before
ar, each run of ar shows its output, and POST runs after, both through the
shell.

```cu
(do (proc-run (list "/bin/sh" "-c" "rm -rf /tmp/x-cu-ar && mkdir -p /tmp/x-cu-ar")) (def ar-f (fn (_ n) (string-append "/tmp/x-cu-ar/" n))) (file-write-all (ar-f "mk.sh") "
umask 022
h() { printf '%-16s%-12s%-6s%-6s%-8s%-10s`\\n' \"$1\" \"$2\" \"$3\" \"$4\" \"$5\" \"$6\"; }
{ printf '!<arch>\\n'; h hello.txt/ 1577934245 1000 1000 100644 6; printf 'hello\\n'
  h odd/ 1577934245 0 0 100755 3; printf 'ab\\n\\n'
  h z.c/ 1577934245 1000 100 100600 0; } > a.a
{ printf '!<arch>\\n'; h / 0 0 0 0 4; printf '\\0\\0\\0\\0'
  h // '' '' '' '' 42; printf 'a-rather-long-name.o/\\nanother-long-one.o/\\n'
  h /0 1577934245 0 0 100644 2; printf 'A\\n'
  h /22 1577934245 0 0 104755 2; printf 'B\\n'; } > gnu.a
printf '!<arcx>\\n' > bad.a
{ printf '!<arch>\\n'; h x/ 0 0 0 100644 2 | sed 's/`$/!/'; printf 'x\\n'; } > badhdr.a
{ printf '!<arch>\\n'; h x/ 0 0 0 100644 12a; printf 'x\\n'; } > badnum.a
{ printf '!<arch>\\n'; h x/ 0 0 0 100648 2; printf 'x\\n'; } > badmode.a
{ printf '!<arch>\\n'; h x/ 0 0 0 100644 10; printf 'abc\\n'; } > short.a
{ printf '!<arch>\\n'; h /5 0 0 0 100644 2; printf 'x\\n'; } > longref.a
{ printf '!<arch>\\n'; h x/ 0 0 0 100644 2; printf 'x\\n'; printf 'tail'; } > tail.a
: > e.a
printf 'not an archive\\n' > text
printf 'one\\n' > f1; printf 'second file\\n' > f2; mkdir d; printf 'deep\\n' > d/inner.txt
printf 'replaced\\n' > hello.txt.new
printf 'L\\n' > a-name-of-twenty-bytes
touch -t 202001020304.05 f1 f2 d/inner.txt a-name-of-twenty-bytes hello.txt.new
touch -t 202001020304.04 ../lo; touch -t 202001020304.06 ../hi
") (file-write-all (ar-f "show.sh") "# the archive A with each header's owner fields shown as OWNER
sed -E 's#^(.{28}).{12}(.{18}`)$#\\1OWNER       \\2#' \"$1\"
") (file-write-all (ar-f "files.sh") "# each FILE named: its mode and its bytes, or that it is not there
for f in \"$@\"; do if [ -f \"$f\" ]; then printf '%s %s: %s\\n' \"$(ls -l \"$f\" | cut -c1-10)\" \"$f\" \"$(cat \"$f\")\"; else printf '%s: none\\n' \"$f\"; fi; done
") (file-write-all (ar-f "when.sh") "# each FILE: whether its time is the archive's (2020-01-02 03:04:05 UTC)
for f in \"$@\"; do if [ \"$f\" -nt ../lo ] && [ \"$f\" -ot ../hi ]; then echo \"$f: the member's time\"; else echo \"$f: now\"; fi; done
") (def ar-home (Sys getcwd)) (def ar-sh (fn (_ cmd out) (proc-run (list "/bin/sh" "-c" (%str-concat (list "TZ=UTC; export TZ; cd /tmp/x-cu-ar/w && { " cmd "; } > " out " 2>&1")))))) (def ar-run (fn (_ argv) (do (sys-dup2 1 9) (sys-dup2 2 8) (let ((oo (file-open-write (ar-f "o"))) (ee (file-open-write (ar-f "e")))) (do (sys-dup2 oo 1) (sys-dup2 ee 2) (def rs (cu-run argv "")) (sys-dup2 9 1) (sys-dup2 8 2) (file-close oo) (file-close ee) (ar-sh "sed -E 's#[A-Z][a-z]{2} [ 0-9][0-9] [0-9]{2}:[0-9]{2} [0-9]{4}#TIME#; s/$/|/' ../o" (ar-f "o.n")) (%str-concat (list (file-read-all (ar-f "o.n")) "stderr:\n" (file-read-all (ar-f "e")) "status " (cu-int->str rs) "\n"))))))) (def ar-case (fn (_ pre post runs) (do (proc-run (list "/bin/sh" "-c" "TZ=UTC; export TZ; cd /tmp/x-cu-ar && rm -rf w && mkdir w && cd w && sh ../mk.sh")) (if pre (ar-sh pre "/dev/null") ()) (Sys chdir "/tmp/x-cu-ar/w") (def shown (%str-concat (map ar-run runs))) (Sys chdir ar-home) (display (string-append shown (if post (do (ar-sh post (ar-f "p")) (string-append "after:\n" (file-read-all (ar-f "p")))) "")))))) (display "made"))
```
